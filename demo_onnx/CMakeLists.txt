cmake_minimum_required(VERSION 3.20)
find_package(Python 3.10 EXACT COMPONENTS Interpreter Development.Module REQUIRED)
if(CMAKE_CROSSCOMPILING)
  # The robot toolchain provides the target SOABI; FindPython's interpreter
  # runs on the build host, while Python_INCLUDE_DIR must select target headers.
  if(NOT DEFINED PYTHON_SOABI OR NOT Python_INCLUDE_DIR)
    message(FATAL_ERROR "Cross build requires target PYTHON_SOABI and Python_INCLUDE_DIR")
  endif()
  set(Python_SOABI "${PYTHON_SOABI}")
endif()
find_package(pybind11 CONFIG REQUIRED NO_CMAKE_FIND_ROOT_PATH)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Only Linux x86_64 and aarch64 wheels are supported")
endif()
set(ORT_ARCHIVE "" CACHE FILEPATH "Pinned official ONNX Runtime CPU tarball")
if(NOT EXISTS "${ORT_ARCHIVE}")
  message(FATAL_ERROR
    "ORT_ARCHIVE is missing. From the SDK root, download and SHA-verify the pinned archive:\n"
    "  python tools/fetch_ort.py --output /absolute/path/to/onnxruntime.tgz\n"
    "Then pass its path for either editable installation or wheel construction:\n"
    "  CMAKE_ARGS='-DORT_ARCHIVE=/absolute/path/to/onnxruntime.tgz' python -m pip install -e .\n"
    "  CMAKE_ARGS='-DORT_ARCHIVE=/absolute/path/to/onnxruntime.tgz' make build\n"
    "See README.md Development and offline packaging for build dependencies.")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ort-archives.json" manifest)
string(JSON ort_version GET "${manifest}" version)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|aarch64)$")
  message(FATAL_ERROR "Unsupported target CPU: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
string(JSON expected_sha GET "${manifest}" "${CMAKE_SYSTEM_PROCESSOR}" sha256)
string(JSON archive_directory GET "${manifest}" "${CMAKE_SYSTEM_PROCESSOR}" directory)
file(SHA256 "${ORT_ARCHIVE}" actual_sha)
if(NOT actual_sha STREQUAL expected_sha)
  message(FATAL_ERROR "ORT archive SHA256 mismatch for ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(vendor "${CMAKE_CURRENT_BINARY_DIR}/vendor")
file(ARCHIVE_EXTRACT INPUT "${ORT_ARCHIVE}" DESTINATION "${vendor}")
set(ort_root "${vendor}/${archive_directory}")
add_library(ort SHARED IMPORTED)
set_target_properties(ort PROPERTIES
  IMPORTED_LOCATION "${ort_root}/lib/libonnxruntime.so.${ort_version}"
  INTERFACE_INCLUDE_DIRECTORIES "${ort_root}/include")
# Public native header types/functions must come from the matching Core output.
# Older vendored headers cannot stand in for the fixed bee contract.
set(AORTA_NATIVE_HEADER_INCLUDE_DIR "" CACHE PATH "Directory containing the pinned generated aorta.h")
if(NOT EXISTS "${AORTA_NATIVE_HEADER_INCLUDE_DIR}/aorta.h")
  message(FATAL_ERROR "Real bee-generated aorta.h required: set AORTA_NATIVE_HEADER_INCLUDE_DIR")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../compatibility.json" compatibility_manifest)
string(JSON expected_native_header_sha GET "${compatibility_manifest}" aorta_native_header_sha256)
file(SHA256 "${AORTA_NATIVE_HEADER_INCLUDE_DIR}/aorta.h" actual_native_header_sha)
if(NOT actual_native_header_sha STREQUAL expected_native_header_sha)
  message(FATAL_ERROR "aorta.h differs from the pinned runtime generation")
endif()
pybind11_add_module(_body_cmd_native NO_EXTRAS WITH_SOABI
  cpp/body_cmd_encoder.cpp cpp/body_state_decoder.cpp cpp/end_effector_decoder.cpp)
target_include_directories(_body_cmd_native PRIVATE "${AORTA_NATIVE_HEADER_INCLUDE_DIR}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/cpp/body_cmd_vendor/provenance.json" body_manifest)
set(schema_vendor "${CMAKE_CURRENT_SOURCE_DIR}/cpp/body_cmd_vendor")
string(JSON include_count LENGTH "${body_manifest}" generated_include_directories)
math(EXPR include_last "${include_count} - 1")
foreach(index RANGE ${include_last})
  string(JSON include_path GET "${body_manifest}" generated_include_directories ${index})
  target_include_directories(_body_cmd_native PRIVATE "${schema_vendor}/${include_path}")
endforeach()
string(JSON header_count LENGTH "${body_manifest}" generated_headers_sha256)
math(EXPR header_last "${header_count} - 1")
foreach(index RANGE ${header_last})
  string(JSON header_path MEMBER "${body_manifest}" generated_headers_sha256 ${index})
  string(JSON expected_header_sha GET "${body_manifest}" generated_headers_sha256 "${header_path}")
  file(SHA256 "${schema_vendor}/${header_path}" actual_header_sha)
  if(NOT actual_header_sha STREQUAL expected_header_sha)
    message(FATAL_ERROR "${header_path} differs from its pinned generated bytes")
  endif()
endforeach()
string(JSON body_cmd_sha GET "${body_manifest}" body_cmd_bfbs_sha256)
string(JSON body_state_sha GET "${body_manifest}" body_state_bfbs_sha256)
foreach(kind IN ITEMS cmd state)
  string(JSON expected_body_sha GET "${compatibility_manifest}" schema_sha256 "body_${kind}")
  if(NOT "${body_${kind}_sha}" STREQUAL expected_body_sha)
    message(FATAL_ERROR "Body ${kind} native schema differs from compatibility.json")
  endif()
endforeach()
set(ee_vendor "${CMAKE_CURRENT_SOURCE_DIR}/cpp/end_effector_vendor")
file(READ "${ee_vendor}/provenance.json" ee_manifest)
foreach(kind IN ITEMS cmd state touch)
  set(stem "end_effector_${kind}")
  string(JSON ee_${kind}_sha GET "${ee_manifest}" bfbs_sha256 "${stem}")
  string(JSON expected_schema_sha GET "${compatibility_manifest}" schema_sha256 "${stem}")
  if(NOT "${ee_${kind}_sha}" STREQUAL expected_schema_sha)
    message(FATAL_ERROR "${stem} native schema differs from compatibility.json")
  endif()
endforeach()
target_compile_definitions(_body_cmd_native PRIVATE
  BODY_CMD_SCHEMA_SHA256="${body_cmd_sha}"
  BODY_STATE_SCHEMA_SHA256="${body_state_sha}"
  EE_CMD_SCHEMA_SHA256="${ee_cmd_sha}"
  EE_STATE_SCHEMA_SHA256="${ee_state_sha}"
  EE_TOUCH_SCHEMA_SHA256="${ee_touch_sha}")
target_compile_features(_body_cmd_native PRIVATE cxx_std_17)
target_compile_options(_body_cmd_native PRIVATE "$<$<CONFIG:Release>:-O3;-g;-DNDEBUG>")
install(TARGETS _body_cmd_native LIBRARY DESTINATION locomotion_aorta)
install(FILES cpp/body_cmd_vendor/LICENSE DESTINATION locomotion_aorta/licenses/flatbuffers)
pybind11_add_module(_native NO_EXTRAS WITH_SOABI cpp/session.cpp)
target_compile_features(_native PRIVATE cxx_std_17)
target_compile_options(_native PRIVATE "$<$<CONFIG:Release>:-O3;-g;-DNDEBUG>")
target_link_options(_native PRIVATE -Wl,--disable-new-dtags)
target_link_libraries(_native PRIVATE ort)
set_target_properties(_native PROPERTIES INSTALL_RPATH "$ORIGIN/.libs")
set(destination "locomotion_aorta/_onnx")
install(TARGETS _native LIBRARY DESTINATION "${destination}")
# Real SONAME bytes avoid depending on wheel installers preserving symlinks.
configure_file("${ort_root}/lib/libonnxruntime.so.${ort_version}"
  "${CMAKE_CURRENT_BINARY_DIR}/libonnxruntime.so.1" COPYONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libonnxruntime.so.1"
  "${ort_root}/lib/libonnxruntime_providers_shared.so"
  DESTINATION "${destination}/.libs")
install(FILES "${ort_root}/LICENSE" "${ort_root}/ThirdPartyNotices.txt"
  DESTINATION "${destination}/licenses/onnxruntime")
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/licenses/pybind11/LICENSE"
  DESTINATION "${destination}/licenses/pybind11")
configure_file(build-info.json.in build-info.json @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/build-info.json"
  DESTINATION "${destination}")
